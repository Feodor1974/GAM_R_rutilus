def get_scalar(x):
    if x is None:
        return np.nan
    if isinstance(x, (int, float, np.number)):
        return float(x)
    if isinstance(x, (list, tuple, np.ndarray)):
        return float(x[0]) if len(x) > 0 else np.nan
    if isinstance(x, dict):
        return float(list(x.values())[0]) if x else np.nan
    return np.nan

def total_edof(gam):
    ed = gam.statistics_.get('edof', np.nan)
    if isinstance(ed, (list, tuple, np.ndarray)):
        return float(np.nansum(ed))
    return float(get_scalar(ed))

def robust_fit(gam_class, X, y, **kwargs):
    try:
        gam = gam_class(**kwargs)
        gam.fit(X, y)
        return gam
    except Exception as e:
        print(f"    [Fit failed] {e}")
        return None

def extract_term_pvals(gam):
    pv = gam.statistics_.get('p_values', None)
    if pv is None:
        return [np.nan]
    if isinstance(pv, (list, tuple, np.ndarray)):
        return [float(v) if v is not None else np.nan for v in pv]
    if isinstance(pv, dict):
        return [float(v) if v is not None else np.nan for v in pv.values()]
    return [float(pv)]

def lrt_gaussian_rss(gam_full, y, X):
    try:
        y_pred = gam_full.predict(X)
        rss_full = np.sum((y - y_pred) ** 2)
        rss_null = np.sum((y - np.mean(y)) ** 2)
        lr = rss_null - rss_full
        df_diff = max(total_edof(gam_full) - 1.0, 1.0)
        if lr <= 0:
            return 1.0, lr, df_diff
        return 1 - chi2.cdf(lr, df_diff), lr, df_diff
    except Exception:
        return np.nan, np.nan, np.nan

def lrt_nested_rss(gam_red, gam_full, y, X_red, X_full):
    try:
        rss_red  = np.sum((y - gam_red.predict(X_red)) ** 2)
        rss_full = np.sum((y - gam_full.predict(X_full)) ** 2)
        lr = rss_red - rss_full
        df_diff = max(total_edof(gam_full) - total_edof(gam_red), 1.0)
        if lr <= 0:
            return 1.0, lr, df_diff
        return 1 - chi2.cdf(lr, df_diff), lr, df_diff
    except Exception:
        return np.nan, np.nan, np.nan

def bootstrap_sd_ci(values, n_boot=N_BOOT):
    """Percentile bootstrap CI for SD."""
    if len(values) < 3:
        return np.nan, np.nan
    boot_sds = []
    for _ in range(n_boot):
        sample = np.random.choice(values, size=len(values), replace=True)
        boot_sds.append(np.std(sample, ddof=1))
    return np.percentile(boot_sds, [2.5, 97.5])

responses_all = ['log_k', 'log_L_lim', 'L_max', 'A_i', 'LS']

for driver in drivers_3:
    for response in responses_all:
        df_sorted = df.sort_values(by=driver).reset_index(drop=True)
        
        # --- Default binning ---
        for n_bins in range(default_n_bins, 2, -1):
            try:
                df_sorted['bin'] = pd.qcut(df_sorted[driver], q=n_bins, labels=False, duplicates='drop')
                if df_sorted['bin'].nunique() >= 3:
                    break
            except ValueError:
                continue
        else:
            print(f"  {response} ~ {driver}: cannot create bins")
            continue
        
        bin_stats = []
        for b in sorted(df_sorted['bin'].unique()):
            subset = df_sorted[df_sorted['bin'] == b]
            vals = subset[response].values
            if len(vals) < 3:
                continue
            sd_val = np.std(vals, ddof=1)
            mean_val = np.mean(vals)
            ci_low, ci_high = bootstrap_sd_ci(vals)
            bin_stats.append({
                'Driver': driver, 'Response': response, 'bin_id': b,
                'driver_mid': subset[driver].mean(),
                'driver_min': subset[driver].min(), 'driver_max': subset[driver].max(),
                'sd': sd_val, 'log_sd': np.log(sd_val) if sd_val > 0 else np.nan,
                'cv': sd_val / abs(mean_val) if abs(mean_val) > 1e-9 else np.nan,
                'iqr': np.percentile(vals, 75) - np.percentile(vals, 25),
                'mean_resp': mean_val, 'n_bin': len(vals),
                'sd_ci_low': ci_low, 'sd_ci_high': ci_high
            })
        
        df_bin = pd.DataFrame(bin_stats).dropna()
        if len(df_bin) < 3:
            print(f"  {response} ~ {driver}: only {len(df_bin)} bins")
            continue
        
        # Spearman trend (primary inference)
        rho_sd, p_sd = spearmanr(df_bin['driver_mid'], df_bin['sd'])
        rho_iqr, p_iqr = spearmanr(df_bin['driver_mid'], df_bin['iqr'])
        rho_cv, p_cv = spearmanr(df_bin['driver_mid'], df_bin['cv'])
        
        trend_results.append({
            'Task': 'T6a_Bin_Dispersion',
            'Response': response, 'Driver': driver,
            'n_bins': len(df_bin),
            'rho_sd': rho_sd, 'p_sd': p_sd,
            'rho_iqr': rho_iqr, 'p_iqr': p_iqr,
            'rho_cv': rho_cv, 'p_cv': p_cv,
            'note': 'Marginal; drivers correlated, see T6b'
        })
        print(f"  {response:12s} ~ {driver:12s} | bins={len(df_bin)} | "
              f"SD rho={rho_sd:+.2f} p={p_sd:.3f} | IQR rho={rho_iqr:+.2f} p={p_iqr:.3f}")
        
        bin_data.extend(bin_stats)
        
        # --- Sensitivity: Spearman for n_bins = 3,4,5,6 ---
        for n_test in [3, 4, 5, 6]:
            if n_test > n_total // 5:
                continue
            try:
                df_sorted['bin_test'] = pd.qcut(df_sorted[driver], q=n_test, labels=False, duplicates='drop')
                if df_sorted['bin_test'].nunique() < 3:
                    continue
                bs = []
                for b in sorted(df_sorted['bin_test'].unique()):
                    ss = df_sorted[df_sorted['bin_test'] == b]
                    vv = ss[response].values
                    if len(vv) < 3:
                        continue
                    bs.append({'mid': ss[driver].mean(), 'sd': np.std(vv, ddof=1)})
                df_bs = pd.DataFrame(bs).dropna()
                if len(df_bs) >= 3:
                    r, p = spearmanr(df_bs['mid'], df_bs['sd'])
                else:
                    r, p = np.nan, np.nan
            except Exception:
                r, p = np.nan, np.nan
            
            sensitivity_results.append({
                'Response': response, 'Driver': driver,
                'n_bins_test': n_test, 'rho_sd': r, 'p_sd': p
            })

df_bin_data = pd.DataFrame(bin_data)

resid_results = []

for response in responses_all:
    # Base model
    if response == 'log_k':
        preds_base = eco
        terms_base = s(0, n_splines=N_SPLINES) + s(1, n_splines=N_SPLINES) + s(2, n_splines=N_SPLINES)
    else:
        preds_base = eco + ['log_k']
        terms_base = (s(0, n_splines=N_SPLINES) + s(1, n_splines=N_SPLINES) +
                      s(2, n_splines=N_SPLINES) + s(3, n_splines=N_SPLINES))
    
    sub = df[[response] + preds_base].dropna()
    X_base = sub[preds_base].values
    y_base = sub[response].values
    gam_base = robust_fit(LinearGAM, X_base, y_base, terms=terms_base)
    if gam_base is None:
        continue
    
    y_pred = gam_base.predict(X_base)
    abs_resid = np.abs(y_base - y_pred)
    
    # Full variance model
    X_env = sub[eco].values
    gam_resid = robust_fit(LinearGAM, X_env, abs_resid,
                           terms=(s(0, n_splines=N_SPLINES) + s(1, n_splines=N_SPLINES) + s(2, n_splines=N_SPLINES)))
    # Null variance model
    gam_resid_null = robust_fit(LinearGAM, np.ones((len(sub), 1)), abs_resid, terms=None)
    
    if gam_resid is None:
        continue
    
    aic_full = get_scalar(gam_resid.statistics_.get('AIC', np.nan))
    aic_null = get_scalar(gam_resid_null.statistics_.get('AIC', np.nan)) if gam_resid_null is not None else np.nan
    delta_aic = aic_null - aic_full if not np.isnan(aic_null) else np.nan
    
    p_vals = extract_term_pvals(gam_resid)
    p_ind = {d: (p_vals[i] if i < len(p_vals) else np.nan) for i, d in enumerate(eco)}
    r2 = get_scalar(gam_resid.statistics_.get('pseudo_r2', np.nan))
    edf = total_edof(gam_resid)
    
    # Evidence level
    if np.isnan(delta_aic):
        evidence = 'NA'
    elif delta_aic > 10:
        evidence = 'very strong'
    elif delta_aic > 6:
        evidence = 'strong'
    elif delta_aic > 2:
        evidence = 'moderate'
    else:
        evidence = 'weak/none'
    
    resid_results.append({
        'Task': 'T6b_Heteroscedasticity',
        'Response': response,
        'p_GDD': p_ind['GDD'], 'p_log_Area': p_ind['log_Area'], 'p_AvgDepth': p_ind['AvgDepth'],
        'AIC_full': aic_full, 'AIC_null': aic_null, 'delta_AIC': delta_aic,
        'evidence': evidence,
        'EDF': edf, 'R_squared': r2, 'n': len(sub)
    })
    print(f"  {response:12s} | GDD p={p_ind['GDD']:.3f}, log_Area p={p_ind['log_Area']:.3f}, "
          f"AvgDepth p={p_ind['AvgDepth']:.3f} | ΔAIC={delta_aic:+.1f} ({evidence})")

df_resid = pd.DataFrame(resid_results)

    pivot = sub.pivot(index='Response', columns='n_bins_test', values='rho_sd')
    
    im = ax.imshow(pivot.values, cmap='RdBu_r', aspect='auto', vmin=-1, vmax=1)
    ax.set_xticks(range(len(pivot.columns)))
    ax.set_xticklabels([f'{c} bins' for c in pivot.columns])
    ax.set_yticks(range(len(pivot.index)))
    ax.set_yticklabels(pivot.index)
    ax.set_title(driver, fontsize=12, fontweight='bold')
    
    for i in range(len(pivot.index)):
        for k in range(len(pivot.columns)):
            val = pivot.iloc[i, k]
            pval = df_sensitivity[(df_sensitivity['Driver']==driver)&
                                  (df_sensitivity['Response']==pivot.index[i])&
                                  (df_sensitivity['n_bins_test']==pivot.columns[k])]['p_sd'].values
            if len(pval) > 0 and not np.isnan(val):
                sig = '*' if pval[0] < 0.05 else ''
                text = f'{val:+.2f}{sig}'
                ax.text(k, i, text, ha='center', va='center',
                       color='white' if abs(val) > 0.5 else 'black', fontsize=9)
    fig.colorbar(im, ax=ax, label='Spearman ρ')
fig.suptitle('Sensitivity: Spearman ρ(SD, driver) across bin counts', fontsize=14, y=1.02)
